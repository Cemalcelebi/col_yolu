def cikti_yolu(ad: str) -> str:
    y = os.path.realpath(os.path.join(CIKTI, ad))
    if os.path.commonpath([y, CIKTI]) != CIKTI:
        raise ValueError(f"çıktı yolu K57 klasörü dışında: {y}")
    return y


def gunluk_oku(yollar):
    """label -> son sonuç (aynı etiket yeniden koşulduysa sonuncusu)."""
    out = {}
    for y in yollar:
        etiket = {}
        for satir in open(y, encoding="utf-8"):
            d = json.loads(satir)
            if d.get("type") == "started":
                etiket[d["agentId"]] = d.get("label")
            elif d.get("type") == "result" and d.get("result") is not None and etiket.get(d["agentId"]):
                out[etiket[d["agentId"]]] = d["result"]
    return out


def manifest():
    with open(os.path.join(CIKTI, "KAYNAK_MANIFEST.csv"), encoding="utf-8") as fh:
        return {r["yol"]: r for r in csv.DictReader(fh)}


def hash_bagla(metin: str, man) -> str:
    """Ajanın yazdığı dosya yollarını manifestle eşleştirir; hash'i LLM değil manifest üretir."""
    out = []
    for parca in [p.strip() for p in metin.replace(";", ",").replace(" + ", ",").split(",")]:
        for yol, r in man.items():
            if parca and (yol == parca or yol.endswith("/" + parca) or parca.endswith(yol) or os.path.basename(yol) == os.path.basename(parca.split(" ")[0])):
                if f"{yol}:{r['sha256'][:12]}" not in out:
                    out.append(f"{yol}:{r['sha256'][:12]}")
                break
    return " | ".join(out) if out else "MANİFESTTE EŞLEŞEN DOSYA YOK"

        for t in testler:
